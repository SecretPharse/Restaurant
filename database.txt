TABLE VERIFICATIONS:
ID INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
EMAIL TEXT NOT NULL UNIQUE,
CODE TEXT NOT NULL,
CODE_EXPIRE TIMESTAMPTZ NOT NULL,
VERIFICATION_EXPIRE TIMESTAMPTZ NOT NULL,
COUNT INTEGER NOT NULL DEFAULT 1

TABLE ACCOUNTS:
ID INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
NAME TEXT NOT NULL,
EMAIL TEXT NOT NULL,
PHONE TEXT NOT NULL,
ADDRESS TEXT NOT NULL,
OPEN TIME NOT NULL,
CLOSE TIME NOT NULL,
WORKER INTEGER NOT NULL DEFAULT 0,
TABLE INTEGER NOT NULL DEFAULT 10,
PASSWORD TEXT NOT NULL,

FOREIGN KEY (EMAIL),
    REFERENCES VERIFICATIONS(EMAIL)
